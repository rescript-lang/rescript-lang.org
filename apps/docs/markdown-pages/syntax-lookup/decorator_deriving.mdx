Use `@deriving(accessors)` on a record type to generate functions that read its fields.

The old `@deriving(abstract)` form was removed in v13. Use ordinary records and record literals for JavaScript objects, including optional fields and mutable fields when needed.

### Example

<CodeTab labels={["ReScript", "JS Output"]}>

```res
@deriving(accessors)
type person = {
  name: string,
  age: int,
  job: string,
}

let joe = {name: "Joe", age: 20, job: "teacher"}

let joeName = name(joe)
let joeAge = age(joe)
let joeJob = job(joe)
```

```js
function name(param) {
  return param.name;
}

function age(param) {
  return param.age;
}

function job(param) {
  return param.job;
}

let joe = { name: "Joe", age: 20, job: "teacher" };

let joeName = "Joe";

let joeAge = 20;

let joeJob = "teacher";

export { name, age, job, joe, joeName, joeAge, joeJob };
```

</CodeTab>
