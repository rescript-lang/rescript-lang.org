Use `module type` to declare a module signature.

### Example

<CodeTab labels={["ReScript", "JS Output"]}>

```res
module type Showable = {
  type t
  let show: t => string
}

module IntShow: Showable with type t = int = {
  type t = int
  let show = x => Int.toString(x)
}

let rendered = IntShow.show(42)
```

```js
function show(x) {
  return x.toString();
}

let IntShow = { show: show };

let rendered = (42).toString();

export { IntShow, rendered };
```

</CodeTab>

### Type Constraints

Use `with type` to refine an abstract type from another module type. A `private` constraint exposes the underlying representation for safe coercion without allowing callers to construct values directly:

```res
module type Storage = {
  type t
  let make: string => t
}

module type PrivateStringStorage = Storage with type t = private string
```
