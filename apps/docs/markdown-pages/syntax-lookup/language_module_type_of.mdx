Use `module type of` when you want a signature based on an existing module.

### Example

<CodeTab labels={["ReScript", "JS Output"]}>

```res
module Message = {
  let greeting = "Hello"
  let prefix = "[msg]"
}

module type MessageShape = module type of Message

module LoudMessage: MessageShape = {
  let greeting = "HELLO"
  let prefix = "[LOUD]"
}
```

```js
let Message = { greeting: "Hello", prefix: "[msg]" };

let LoudMessage = { greeting: "HELLO", prefix: "[LOUD]" };

export { Message, LoudMessage };
```

</CodeTab>
